Atualiza uma automação existente. O corpo substitui a automação inteira; campos omitidos podem voltar vazios.

## Parâmetros de rota

<Params>
<Param name="id" type="int">
ID da automação a atualizar.
</Param>
</Params>

## Corpo da requisição

<Params>
<Param name="name" type="string" required>
Nome da automação
</Param>
<Param name="description" type="string" required={false}>
Descrição
</Param>
<Param name="scope" type="int" required>
Escopo da automação
</Param>
<Param name="triggerType" type="int" required>
Tipo de gatilho
</Param>
<Param name="triggerConfig" type="object" required>
Configuração do gatilho
</Param>
<Param name="conditions" type="array" required>
Array de condições
</Param>
<Param name="actions" type="array" required>
Array de ações
</Param>
<Param name="cooldownMinutes" type="int" required={false}>
Intervalo mínimo entre execuções
</Param>
<Param name="maxExecutionsPerDay" type="int" required={false}>
Limite de execuções por dia
</Param>
<Param name="maxExecutionsPerEntity" type="int" required={false}>
Limite de execuções por entidade
</Param>
<Param name="funnelId" type="int" required={false}>
ID do funil
</Param>
<Param name="departmentId" type="int" required={false}>
ID do departamento
</Param>
<Param name="isActive" type="bool" required={false}>
Se a automação está ativa
</Param>
</Params>
